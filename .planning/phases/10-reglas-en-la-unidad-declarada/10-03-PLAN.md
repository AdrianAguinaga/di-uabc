    d = copy.deepcopy(CURSO_VALIDO)
    next(r for r in d["evaluacion"]["rubros"] if r["id"] == "examenes")["parciales"] = 3
    d["metas"] = [m for m in d["metas"] if m["id"] not in ("P1", "P2")]
