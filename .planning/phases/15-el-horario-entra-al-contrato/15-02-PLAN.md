        """Fecha del día `dia` (0=lunes … 5=sábado) de la semana indicada.

```python
    def dia_de_clase(self, semana: int, dia: int, dias_con_clase: set[int]) -> date | None:
        """El primer día de esa semana, desde `dia`, sin suspensión y con clase para el grupo.

        `dias_con_clase` son los días (0=lunes … 5=sábado) en los que el grupo tiene bloque
        presencial. El recorrido **no sale de la semana**: si la suspensión deja al grupo sin
        ningún día con clase, devuelve `None`. Aterrizar en la semana siguiente imprimiría en
        la fila «semana 13» una fecha de la semana 14 (D-15); quien pregunta decide qué hacer.
        """
        s = self.semana(semana)
        objetivo = s.inicio + timedelta(days=dia)
        while objetivo <= s.fin:
            if objetivo.weekday() in dias_con_clase and not self.es_suspension(objetivo):
                return objetivo
            objetivo += timedelta(days=1)
        return None
```

```python
    def fecha_de(
        self, semana: int, dia: int = 0, dias_con_clase: set[int] | None = None
    ) -> date:
        """Fecha del día `dia` (0=lunes … 5=sábado) de la semana indicada.

        Sin `dias_con_clase`, una suspensión se recorre al siguiente día hábil, como siempre.
        Con él, el recorrido solo acepta días en los que el grupo tiene bloque presencial y no
        sale de la semana: si ninguno sirve, devuelve el día original —aunque sea suspensión— y
        `validar.py` lo reporta (D-07/D-15). Un grupo sin bloques nunca pasa por esta rama.
        """
        s = self.semana(semana)
        objetivo = s.inicio + timedelta(days=dia)
        if dias_con_clase is not None:
            objetivo = self.dia_de_clase(semana, dia, dias_con_clase) or objetivo
        else:
            while self.es_suspension(objetivo):
                objetivo += timedelta(days=1)
        if objetivo > self.fin:
            raise ErrorCalendario(
                f"La fecha calculada ({objetivo}) queda después del fin de cursos "
                f"({self.fin}). Adelanta la entrega."
            )
        return objetivo
```
