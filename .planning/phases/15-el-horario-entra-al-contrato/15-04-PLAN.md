    def _horario(self) -> None:
        """La forma del horario del grupo: horas legibles, coherentes y sin cruzarse (D-05).

        Vive dentro de R6 —el dominio «calendario y fechas»— y no en una regla nueva: es el
        mismo trato que la Fase 12 dio a la rúbrica dentro de R2. Que la hora sea legible es
        hallazgo y no `ErrorModelo` a propósito: el criterio 3 pide que lo reporte
        `python src/validar.py`, con el resto de los hallazgos y sin reventar la carga.
        """
        for g in self.c.grupos:
            tramos: dict[int, list[tuple[time, time]]] = {}
            for b in g.horario.bloques:
                dia = calendario.DIAS[b.dia]
                horas: dict[str, time] = {}
                for campo, valor in (("inicio", b.inicio), ("fin", b.fin)):
                    try:
                        horas[campo] = time.fromisoformat(str(valor))
                    except ValueError:
                        self.error(
                            "R6",
                            f"Grupo {g.numero}, bloque del {dia}: la hora de {campo} "
                            f"«{valor}» no está escrita como HH:MM.",
                        )
                if len(horas) < 2:
                    continue
                if horas["fin"] <= horas["inicio"]:
                    self.error(
                        "R6",
                        f"Grupo {g.numero}, bloque del {dia}: la hora de fin ({b.fin}) no es "
                        f"posterior a la de inicio ({b.inicio}).",
                    )
                    continue
                tramos.setdefault(b.dia, []).append((horas["inicio"], horas["fin"]))

            for dia, lista in tramos.items():
                lista.sort()
                for (ini, fin), (sig_ini, sig_fin) in zip(lista, lista[1:]):
                    if sig_ini < fin:
                        self.error(
                            "R6",
                            f"Grupo {g.numero}: dos bloques del {calendario.DIAS[dia]} se "
                            f"solapan ({ini:%H:%M}–{fin:%H:%M} y "
                            f"{sig_ini:%H:%M}–{sig_fin:%H:%M}); un grupo no puede estar en dos "
                            f"clases a la vez.",
                        )

    def _semanas_sin_dia_de_clase(self) -> None:
        """La suspensión que deja al grupo sin ningún día de clase en esa semana (D-15).

        Se computa con **horario + calendario** y nada más: `generar.paquete()` valida antes de
        que `render_docx` resuelva las fechas, así que un hallazgo que leyera `Sesion.fecha` no
        se dispararía nunca en el pipeline real. Pregunta a la misma función que resuelve la
        fecha —`dia_de_clase`—, de modo que el aviso y la fecha impresa no puedan discrepar.

        Es aviso y no error: la clase existe y el calendario la suspendió. Reprogramarla es
        decisión del docente (Art. 66), y bloquear la generación dejaría al curso sin documento
        por una circunstancia del calendario oficial.
        """
        for g in self.c.grupos:
            if not g.imparte or not g.horario.bloques or not g.horario.dias_presencial:
                continue
            dias = set(g.horario.dias_presencial)
            primero = g.horario.dias_presencial[0]

            for n in range(1, self.cal.total_semanas + 1):
                if self.cal.dia_de_clase(n, primero, dias) is not None:
                    continue
                suspendidas = ", ".join(
                    calendario.texto_fecha(s.fecha) for s in self.cal.semana(n).suspensiones
                )
                self.aviso(
                    "R6",
                    f"Semana {n}: el grupo {g.numero} no tiene ningún día con bloque "
                    f"presencial ({suspendidas} es suspensión). La fecha de esa semana se "
                    f"queda en el día suspendido; reprograma esa clase si debe darse.",
                )

            for m in self.c.metas:
                for s in m.sesiones:
                    if s.dia is not None and s.ambiente == "presencial" and s.dia not in dias:
                        self.aviso(
                            "R6",
                            f"La {m.etiqueta} fija su sesión presencial en {calendario.DIAS[s.dia]} "
                            f"y el grupo {g.numero} no tiene bloque presencial ese día.",
                        )
